788/*============================================================================
789 * Fast Macro Sprite API
790 *
791 * Zero-overhead alternatives to oamSet/oamSetXY for performance-critical code.
792 * These write directly to oamMemory[] without function call overhead.
793 *
794 * oamSet() is a call with seven arguments pushed and a stack frame; for a
795 * handful of sprites updated every frame the macros below write the four
796 * OAM bytes in place. (A byte count of an earlier oamSet's frame and a
797 * "visible jitter" claim stood here until 2026-10-04; the cost is in
798 * docs/PERF.md, measured, not here.)
799 *
800 * Note: cc65816 does not truly inline 'static inline' functions — they
801 * become separate SUPERFREE sections with global labels that conflict
802 * across translation units. Macros are the only zero-overhead option.
803 *
804 * oamMemory[] and oam_update_flag are declared in <snes/system.h>
805 * (included automatically via <snes.h>).
806 *
807 * Usage:
808 * // Pre-compute attribute byte once at init
809 * u8 attr = OAM_ATTR(tile, palette, priority, flags);
810 *
811 * // Per-frame: fast full update
812 * oamSetFast(id, x, y, tile, palette, priority, flags);
813 *
814 * // Per-frame: position-only update (most common)
815 * oamSetXYFast(id, x, y);
816 *============================================================================*/
817
827#define OAM_ATTR(_tile, _pal, _prio, _fl) \
828 ((u8)(((_fl) & 0xC0) | \
829 (((_prio) & 0x03) << 4) | \
830 (((_pal) & 0x07) << 1) | \
831 (((_tile) >> 8) & 0x01)))
832
839#define OAM_XHI_MASK(_slot) \
840 ((u8)((_slot) == 0 ? 0x01 : (_slot) == 1 ? 0x04 : \
841 (_slot) == 2 ? 0x10 : 0x40))
842
858#define oamSetFast(_id, _x, _y, _tile, _pal, _prio, _fl) do { \
859 u16 _off = (u16)(_id) << 2; \
860 oamMemory[_off + 0] = (u8)((_x) & 0xFF); \
861 oamMemory[_off + 1] = (u8)((_y) & 0xFF); \
862 oamMemory[_off + 2] = (u8)((_tile) & 0xFF); \
863 oamMemory[_off + 3] = OAM_ATTR(_tile, _pal, _prio, _fl); \
864 u16 _ext = 512 + ((u16)(_id) >> 2); \
865 u16 _sl = (u16)(_id) & 0x03; \
866 u8 _xhi = OAM_XHI_MASK(_sl); \
867 if ((_x) & 0x100) \
868 oamMemory[_ext] |= _xhi; \
869 else \
870 oamMemory[_ext] &= ~_xhi; \
871 oam_update_flag = 1; \
872} while(0)
873
884#define oamSetXYFast(_id, _x, _y) do { \
885 u16 _off = (u16)(_id) << 2; \
886 oamMemory[_off + 0] = (u8)((_x) & 0xFF); \
887 oamMemory[_off + 1] = (u8)((_y) & 0xFF); \
888 u16 _ext = 512 + ((u16)(_id) >> 2); \
889 u16 _sl = (u16)(_id) & 0x03; \
890 u8 _xhi = OAM_XHI_MASK(_sl); \
891 if ((_x) & 0x100) \
892 oamMemory[_ext] |= _xhi; \
893 else \
894 oamMemory[_ext] &= ~_xhi; \
895 oam_update_flag = 1; \
896} while(0)
