891/*============================================================================
892 * Fast Macro Sprite API
893 *
894 * Inline alternatives to oamSet/oamSetXY: they write directly to oamMemory[]
895 * without a function call.
896 *
897 * The macros below write the OAM bytes in place, from C. They are NOT
898 * cheaper than the assembly setters: measured per sprite (devtools/libbench,
899 * SlowROM), oamSetFast() 2,590 master cycles against oamSet() 2,030 and
900 * oamSetAttr() 1,580, oamSetXYFast() 1,920 against oamSetXY() 1,330. They
901 * stay for code that wants the writes inline; for speed, call the setters.
902 *
903 * Note: cc65816 does not truly inline 'static inline' functions — they
904 * become separate SUPERFREE sections with global labels that conflict
905 * across translation units; hence macros.
906 *
907 * oamMemory[] and oam_update_flag are declared in <snes/system.h>
908 * (included automatically via <snes.h>).
909 *
910 * Usage:
911 * // Pre-compute attribute byte once at init, then per frame:
912 * u8 attr = OAM_ATTR(tile, palette, priority, flags);
913 * oamSetAttr(id, x, y, tile, attr); // the cheapest full update
914 *
915 * // the inline forms
916 * oamSetFast(id, x, y, tile, palette, priority, flags);
917 * oamSetXYFast(id, x, y);
918 *============================================================================*/
919
929#define OAM_ATTR(_tile, _pal, _prio, _fl) \
930 ((u8)(((_fl) & 0xC0) | \
931 (((_prio) & 0x03) << 4) | \
932 (((_pal) & 0x07) << 1) | \
933 (((_tile) >> 8) & 0x01)))
934
941#define OAM_XHI_MASK(_slot) \
942 ((u8)((_slot) == 0 ? 0x01 : (_slot) == 1 ? 0x04 : \
943 (_slot) == 2 ? 0x10 : 0x40))
944
959#define oamSetFast(_id, _x, _y, _tile, _pal, _prio, _fl) do { \
960 u16 _off = (u16)(_id) << 2; \
961 oamMemory[_off + 0] = (u8)((_x) & 0xFF); \
962 oamMemory[_off + 1] = (u8)((_y) & 0xFF); \
963 oamMemory[_off + 2] = (u8)((_tile) & 0xFF); \
964 oamMemory[_off + 3] = OAM_ATTR(_tile, _pal, _prio, _fl); \
965 u16 _ext = 512 + ((u16)(_id) >> 2); \
966 u16 _sl = (u16)(_id) & 0x03; \
967 u8 _xhi = OAM_XHI_MASK(_sl); \
968 if ((_x) & 0x100) \
969 oamMemory[_ext] |= _xhi; \
970 else \
971 oamMemory[_ext] &= ~_xhi; \
972 oam_update_flag = 1; \
973} while(0)
974
985#define oamSetXYFast(_id, _x, _y) do { \
986 u16 _off = (u16)(_id) << 2; \
987 oamMemory[_off + 0] = (u8)((_x) & 0xFF); \
988 oamMemory[_off + 1] = (u8)((_y) & 0xFF); \
989 u16 _ext = 512 + ((u16)(_id) >> 2); \
990 u16 _sl = (u16)(_id) & 0x03; \
991 u8 _xhi = OAM_XHI_MASK(_sl); \
992 if ((_x) & 0x100) \
993 oamMemory[_ext] |= _xhi; \
994 else \
995 oamMemory[_ext] &= ~_xhi; \
996 oam_update_flag = 1; \
997} while(0)
