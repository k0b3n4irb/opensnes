679/*============================================================================
680 * Fast Macro Sprite API
681 *
682 * Zero-overhead alternatives to oamSet/oamSetXY for performance-critical code.
683 * These write directly to oamMemory[] without function call overhead.
684 *
685 * oamSet() has framesize=158 per call due to SSA temporaries. With >2-3
686 * sprites/frame in the main loop, the stack manipulation causes visible
687 * jitter. These macros eliminate that overhead entirely.
688 *
689 * Note: cc65816 does not truly inline 'static inline' functions — they
690 * become separate SUPERFREE sections with global labels that conflict
691 * across translation units. Macros are the only zero-overhead option.
692 *
693 * oamMemory[] and oam_update_flag are declared in <snes/system.h>
694 * (included automatically via <snes.h>).
695 *
696 * Usage:
697 * // Pre-compute attribute byte once at init
698 * u8 attr = OAM_ATTR(tile, palette, priority, flags);
699 *
700 * // Per-frame: fast full update
701 * oamSetFast(id, x, y, tile, palette, priority, flags);
702 *
703 * // Per-frame: position-only update (most common)
704 * oamSetXYFast(id, x, y);
705 *============================================================================*/
706
716#define OAM_ATTR(_tile, _pal, _prio, _fl) \
717 ((u8)(((_fl) & 0xC0) | \
718 (((_prio) & 0x03) << 4) | \
719 (((_pal) & 0x07) << 1) | \
720 (((_tile) >> 8) & 0x01)))
721
728#define OAM_XHI_MASK(_slot) \
729 ((u8)((_slot) == 0 ? 0x01 : (_slot) == 1 ? 0x04 : \
730 (_slot) == 2 ? 0x10 : 0x40))
731
747#define oamSetFast(_id, _x, _y, _tile, _pal, _prio, _fl) do { \
748 u16 _off = (u16)(_id) << 2; \
749 oamMemory[_off + 0] = (u8)((_x) & 0xFF); \
750 oamMemory[_off + 1] = (u8)(((_y) - 1) & 0xFF); /* compensate +1 PPU scanline quirk */ \
751 oamMemory[_off + 2] = (u8)((_tile) & 0xFF); \
752 oamMemory[_off + 3] = OAM_ATTR(_tile, _pal, _prio, _fl); \
753 u16 _ext = 512 + ((u16)(_id) >> 2); \
754 u16 _sl = (u16)(_id) & 0x03; \
755 u8 _xhi = OAM_XHI_MASK(_sl); \
756 if ((_x) & 0x100) \
757 oamMemory[_ext] |= _xhi; \
758 else \
759 oamMemory[_ext] &= ~_xhi; \
760 oam_update_flag = 1; \
761} while(0)
762
773#define oamSetXYFast(_id, _x, _y) do { \
774 u16 _off = (u16)(_id) << 2; \
775 oamMemory[_off + 0] = (u8)((_x) & 0xFF); \
776 oamMemory[_off + 1] = (u8)(((_y) - 1) & 0xFF); /* compensate +1 PPU scanline quirk */ \
777 u16 _ext = 512 + ((u16)(_id) >> 2); \
778 u16 _sl = (u16)(_id) & 0x03; \
779 u8 _xhi = OAM_XHI_MASK(_sl); \
780 if ((_x) & 0x100) \
781 oamMemory[_ext] |= _xhi; \
782 else \

785} while(0)
