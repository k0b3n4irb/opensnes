727/*============================================================================
728 * Fast Macro Sprite API
729 *
730 * Zero-overhead alternatives to oamSet/oamSetXY for performance-critical code.
731 * These write directly to oamMemory[] without function call overhead.
732 *
733 * oamSet() is a call with seven arguments pushed and a stack frame; for a
734 * handful of sprites updated every frame the macros below write the four
735 * OAM bytes in place. (A byte count of an earlier oamSet's frame and a
736 * "visible jitter" claim stood here until 2026-10-04; the cost is in
737 * docs/PERF.md, measured, not here.)
738 *
739 * Note: cc65816 does not truly inline 'static inline' functions — they
740 * become separate SUPERFREE sections with global labels that conflict
741 * across translation units. Macros are the only zero-overhead option.
742 *
743 * oamMemory[] and oam_update_flag are declared in <snes/system.h>
744 * (included automatically via <snes.h>).
745 *
746 * Usage:
747 * // Pre-compute attribute byte once at init
748 * u8 attr = OAM_ATTR(tile, palette, priority, flags);
749 *
750 * // Per-frame: fast full update
751 * oamSetFast(id, x, y, tile, palette, priority, flags);
752 *
753 * // Per-frame: position-only update (most common)
754 * oamSetXYFast(id, x, y);
755 *============================================================================*/
756
766#define OAM_ATTR(_tile, _pal, _prio, _fl) \
767 ((u8)(((_fl) & 0xC0) | \
768 (((_prio) & 0x03) << 4) | \
769 (((_pal) & 0x07) << 1) | \
770 (((_tile) >> 8) & 0x01)))
771
778#define OAM_XHI_MASK(_slot) \
779 ((u8)((_slot) == 0 ? 0x01 : (_slot) == 1 ? 0x04 : \
780 (_slot) == 2 ? 0x10 : 0x40))
781
797#define oamSetFast(_id, _x, _y, _tile, _pal, _prio, _fl) do { \
798 u16 _off = (u16)(_id) << 2; \
799 oamMemory[_off + 0] = (u8)((_x) & 0xFF); \
800 oamMemory[_off + 1] = (u8)(((_y) - 1) & 0xFF); /* compensate +1 PPU scanline quirk */ \
801 oamMemory[_off + 2] = (u8)((_tile) & 0xFF); \
802 oamMemory[_off + 3] = OAM_ATTR(_tile, _pal, _prio, _fl); \
803 u16 _ext = 512 + ((u16)(_id) >> 2); \
804 u16 _sl = (u16)(_id) & 0x03; \
805 u8 _xhi = OAM_XHI_MASK(_sl); \
806 if ((_x) & 0x100) \
807 oamMemory[_ext] |= _xhi; \
808 else \
809 oamMemory[_ext] &= ~_xhi; \
810 oam_update_flag = 1; \
811} while(0)

826 oamMemory[_off + 1] = (u8)(((_y) - 1) & 0xFF); /* compensate +1 PPU scanline quirk */ \
