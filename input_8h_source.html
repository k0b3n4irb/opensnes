60/*
61 * SNES Joypad Register Layout (16-bit read as JOY1L | JOY1H << 8):
62 *
63 * $4219 (JOY1H) → bits 15-8: B, Y, Select, Start, Up, Down, Left, Right
64 * $4218 (JOY1L) → bits 7-0: A, X, L, R, (signature bits)
65 *
66 * Bit layout (16-bit, active high):
67 * 15 14 13 12 11 10 9 8 7 6 5 4 3-0
68 * B Y Sel Sta Up Dn Lt Rt A X L R ID
69 *
70 * Verified against fullsnes documentation and real hardware.
71 */
72
73/* High byte ($4219 → result bits 15-8) */
74#define KEY_B BIT(15)
75#define KEY_Y BIT(14)
76#define KEY_SELECT BIT(13)
77#define KEY_START BIT(12)
78#define KEY_UP BIT(11)
79#define KEY_DOWN BIT(10)
80#define KEY_LEFT BIT(9)
81#define KEY_RIGHT BIT(8)
82
83/* Low byte ($4218 → result bits 7-0) */
84#define KEY_A BIT(7)
85#define KEY_X BIT(6)
86#define KEY_L BIT(5)
87#define KEY_R BIT(4)
88
90#define KEY_DPAD (KEY_UP | KEY_DOWN | KEY_LEFT | KEY_RIGHT)

148/* padHeld() and padPressed() read one word that the NMI handler filled, so
149 * they are also macros (since 2026-10-09): a call cost about 125 CPU cycles,
150 * three times a frame in most games, for what is one load. The handler
151 * already stores 0 for a port that holds no pad, which is all the functions
152 * added. The functions remain — `(padHeld)(0)` calls one, and so does taking
153 * its address. Unlike them the macros do not check `pad`: an index above 4
154 * reads whatever follows the array. */
155extern u16 pad_keys[5];
156extern u16 pad_keysold[5];
157extern u16 pad_keysdown[5];
158#define padHeld(pad) (pad_keys[(pad)])
159#define padPressed(pad) (pad_keysdown[(pad)])
